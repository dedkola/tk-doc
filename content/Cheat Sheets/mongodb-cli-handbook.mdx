## Installation & Setup

### Install MongoDB

```bash
# Ubuntu / Debian (MongoDB 7.0)
curl -fsSL https://www.mongodb.org/static/pgp/server-7.0.asc | \
  sudo gpg -o /usr/share/keyrings/mongodb-server-7.0.gpg --dearmor
echo "deb [ signed-by=/usr/share/keyrings/mongodb-server-7.0.gpg ] https://repo.mongodb.org/apt/ubuntu jammy/mongodb-org/7.0 multiverse" | \
  sudo tee /etc/apt/sources.list.d/mongodb-org-7.0.list
sudo apt update
sudo apt install -y mongodb-org

# Fedora / RHEL
sudo tee /etc/yum.repos.d/mongodb-org-7.0.repo << 'EOF'
[mongodb-org-7.0]
name=MongoDB Repository
baseurl=https://repo.mongodb.org/yum/redhat/8/mongodb-org/7.0/x86_64/
gpgcheck=1
enabled=1
gpgkey=https://www.mongodb.org/static/pgp/server-7.0.asc
EOF
sudo dnf install -y mongodb-org

# macOS (Homebrew)
brew tap mongodb/brew
brew install mongodb-community@7.0
brew services start mongodb-community@7.0

# Verify
mongosh --version
mongod --version
```

### Start & Connect

```bash
# Linux (systemd)
sudo systemctl start mongod
sudo systemctl enable mongod
sudo systemctl status mongod

# Connect with mongosh (modern shell — replaces mongo)
mongosh                                    # Local default (localhost:27017)
mongosh "mongodb://localhost:27017"
mongosh -u admin -p password --authenticationDatabase admin
mongosh --host replica1:27017,replica2:27017,replica3:27017 --replicaSet myrs
mongosh "mongodb+srv://cluster0.example.mongodb.net/" --apiKey MV3...

# Legacy mongo shell (MongoDB < 5.0)
mongo                                      # Deprecated — use mongosh
```

## CRUD Operations

### Insert

```javascript
// Insert single document
db.users.insertOne({
  name: "Alice",
  email: "alice@example.com",
  age: 30,
  address: { city: "Portland", state: "OR" },
  tags: ["admin", "active"],
  createdAt: new Date()
})

// Insert multiple documents
db.users.insertMany([
  { name: "Bob", email: "bob@example.com", age: 25 },
  { name: "Charlie", email: "charlie@example.com", age: 35 }
])

// Insert with ordered: false (continue on error)
db.users.insertMany([...], { ordered: false })

// Bulk write (insert, update, delete in one operation)
db.users.bulkWrite([
  { insertOne: { document: { name: "Dave" } } },
  { updateOne: { filter: { name: "Alice" }, update: { $set: { age: 31 } } } },
  { deleteOne: { filter: { name: "Bob" } } }
])
```

### Query (Read)

```javascript
// Find all documents
db.users.find()
db.users.find().pretty()

// Find with filter
db.users.find({ age: 30 })
db.users.find({ age: { $gt: 25 } })        // Greater than
db.users.find({ age: { $gte: 25, $lte: 40 } })  // Range
db.users.find({ tags: "admin" })            // Array contains
db.users.find({ "address.city": "Portland" }) // Nested field
db.users.find({ email: { $regex: /@example/ } }) // Regex

// Comparison operators
// $eq, $ne, $gt, $gte, $lt, $lte, $in, $nin
db.users.find({ age: { $in: [25, 30, 35] } })
db.users.find({ tags: { $nin: ["inactive"] } })

// Logical operators
db.users.find({ $and: [{ age: { $gte: 25 } }, { age: { $lte: 40 } }] })
db.users.find({ $or: [{ name: "Alice" }, { name: "Bob" }] })
db.users.find({ $not: { age: { $lt: 25 } } })
db.users.find({ $nor: [{ age: 25 }, { age: 30 }] })

// Projection (select fields)
db.users.find({}, { name: 1, email: 1 })       // Include only name, email
db.users.find({}, { _id: 0, name: 1 })         // Exclude _id
db.users.find({}, { name: 1, email: 1 }).sort({ name: 1 })  // Ascending
db.users.find({}, { name: 1 }).sort({ age: -1 })  // Descending

// Limit and skip
db.users.find().limit(10)
db.users.find().skip(20).limit(10)             // Pagination
db.users.find().sort({ age: 1 }).skip(10).limit(5)

// Count
db.users.countDocuments({ age: { $gte: 25 } })
db.users.estimatedDocumentCount()              // Fast count (no filter)

// Find one
db.users.findOne({ name: "Alice" })
db.users.findOne({ name: "Alice" }, { email: 1 })

// Distinct
db.users.distinct("age")
db.users.distinct("address.city")
```

### Update

```javascript
// Update single document
db.users.updateOne(
  { name: "Alice" },
  { $set: { age: 31, "address.city": "Seattle" } }
)

// Update multiple documents
db.users.updateMany(
  { age: { $lt: 25 } },
  { $set: { status: "young" } }
)

// Replace entire document
db.users.replaceOne(
  { name: "Alice" },
  { name: "Alice", email: "alice@new.com", age: 31 }
)

// Update operators
db.users.updateOne({ name: "Alice" }, { $inc: { age: 1 } })           // Increment
db.users.updateOne({ name: "Alice" }, { $unset: { phone: "" } })      // Remove field
db.users.updateOne({ name: "Alice" }, { $push: { tags: "new" } })     // Add to array
db.users.updateOne({ name: "Alice" }, { $pull: { tags: "old" } })     // Remove from array
db.users.updateOne({ name: "Alice" }, { $addToSet: { tags: "vip" } }) // Add if not exists
db.users.updateOne({ name: "Alice" }, { $rename: { "phone": "mobile" } })  // Rename field
db.users.updateOne({ name: "Alice" }, { $min: { age: 28 } })         // Set if lower
db.users.updateOne({ name: "Alice" }, { $max: { age: 35 } })         // Set if higher

// Upsert (update or insert)
db.users.updateOne(
  { email: "new@example.com" },
  { $set: { name: "New User", age: 28 } },
  { upsert: true }
)

// Find and update (returns old or new document)
db.users.findOneAndUpdate(
  { name: "Alice" },
  { $inc: { age: 1 } },
  { returnDocument: "after" }     // "before" (old) or "after" (new)
)
```

### Delete

```javascript
// Delete single document
db.users.deleteOne({ name: "Alice" })

// Delete multiple documents
db.users.deleteMany({ age: { $lt: 18 } })

// Delete all documents
db.users.deleteMany({})

// Find and delete (returns deleted document)
db.users.findOneAndDelete({ name: "Alice" })

// Delete many and return count
db.users.deleteMany({ status: "inactive" }).deletedCount
```

## Aggregation Framework

### Basic Pipeline

```javascript
// Aggregation pipeline (like SQL GROUP BY)
db.orders.aggregate([
  { $match: { status: "completed" } },          // Filter
  { $unwind: "$items" },                        // Flatten array
  { $group: {
      _id: "$userId",
      totalSpent: { $sum: "$items.price" },
      avgOrderValue: { $avg: "$items.price" },
      orderCount: { $sum: 1 }
  }},
  { $sort: { totalSpent: -1 } },                // Sort
  { $limit: 10 },                               // Limit
  { $project: {                                 // Reshape output
      _id: 0,
      userId: "$_id",
      totalSpent: 1,
      avgOrderValue: { $round: ["$avgOrderValue", 2] }
  }}
])
```

### Common Stages

```javascript
// $match — filter documents
{ $match: { status: "active", age: { $gte: 25 } } }

// $group — group and aggregate
{ $group: {
    _id: "$department",
    count: { $sum: 1 },
    totalSalary: { $sum: "$salary" },
    avgSalary: { $avg: "$salary" },
    maxSalary: { $max: "$salary" },
    minSalary: { $min: "$salary" },
    employees: { $push: "$name" }        // Collect into array
}}

// $project — reshape documents
{ $project: {
    name: 1,
    fullName: { $concat: ["$firstName", " ", "$lastName"] },
    ageInYears: "$age",
    _id: 0
}}

// $addFields — add new fields without removing existing
{ $addFields: {
    fullName: { $concat: ["$firstName", " ", "$lastName"] },
    ageNextYear: { $add: ["$age", 1] }
}}

// $sort — sort results
{ $sort: { createdAt: -1 } }           // -1 = descending, 1 = ascending

// $limit and $skip
{ $limit: 10 }
{ $skip: 20 }

// $unwind — flatten array
{ $unwind: "$tags" }

// $lookup — join with another collection (like SQL JOIN)
{ $lookup: {
    from: "orders",
    localField: "_id",
    foreignField: "userId",
    as: "orders"
}}

// $unwind after lookup to flatten
{ $unwind: "$orders" }

// $out — write results to a new collection
{ $out: "aggregated_results" }

// $merge — merge results into existing collection
{ $merge: { into: "summary", on: "_id", whenMatched: "replace" } }

// $facet — multiple aggregations in parallel
{ $facet: {
    byStatus: [
      { $group: { _id: "$status", count: { $sum: 1 } } }
    ],
    byAgeRange: [
      { $bucket: { groupBy: "$age", boundaries: [0, 25, 35, 50, 100] } }
    ]
}}

// $bucket — group into ranges
{ $bucket: {
    groupBy: "$age",
    boundaries: [0, 25, 35, 50, 100],
    default: "Other",
    output: { count: { $sum: 1 }, names: { $push: "$name" } }
}}

// $text — full-text search
{ $match: { $text: { $search: "keyword" } } }
```

### Accumulator Operators

| Operator | Description |
|----------|-------------|
| `$sum` | Sum of values |
| `$avg` | Average of values |
| `$min` / `$max` | Min / max value |
| `$push` | Push values into array |
| `$addToSet` | Push unique values into array |
| `$first` / `$last` | First / last value in group |
| `$mergeObjects` | Merge documents |
| `$count` | Count of documents |

## Indexes

### Create Indexes

```javascript
// Single field index
db.users.createIndex({ email: 1 })                   // Ascending
db.users.createIndex({ createdAt: -1 })              // Descending

// Compound index
db.users.createIndex({ lastName: 1, firstName: 1 })

// Unique index
db.users.createIndex({ email: 1 }, { unique: true })

// Sparse index (only index documents that have the field)
db.users.createIndex({ phone: 1 }, { sparse: true })

// TTL index (auto-delete documents after time)
db.sessions.createIndex({ createdAt: 1 }, { expireAfterSeconds: 3600 })

// Text index (full-text search)
db.articles.createIndex({ title: "text", content: "text" })

// Hashed index (for hash-based sharding)
db.users.createIndex({ _id: "hashed" })

// Wildcard index (index all fields in subdocuments)
db.products.createIndex({ "attributes.$**": 1 })

// Background index (don't block other ops — MongoDB 4.2+ builds in background automatically)
db.users.createIndex({ email: 1 }, { background: true })

// Partial index (only index documents matching filter)
db.orders.createIndex(
  { status: 1 },
  { partialFilterExpression: { status: "active" } }
)

// Create multiple indexes
db.users.createIndexes([
  { email: 1 },
  { name: 1, age: -1 }
])
```

### Manage Indexes

```javascript
// List all indexes
db.users.getIndexes()
db.users.listIndexes()

// Drop index
db.users.dropIndex({ email: 1 })
db.users.dropIndex("email_1")              // By index name
db.users.dropIndexes()                     // Drop all (except _id)
db.users.dropIndexes(["email_1", "name_1"])  // Drop specific indexes

// Analyze index usage
db.users.aggregate([{ $indexStats: {} }])

// Check index size
db.users.stats().indexSizes
```

### Explain Queries

```javascript
// Basic explain
db.users.find({ email: "alice@example.com" }).explain()

// Explain with details
db.users.find({ email: "alice@example.com" }).explain("executionStats")

// Full analyze
db.users.find({ email: "alice@example.com" }).explain("allPlansExecution")

// Key explain output fields:
// winningPlan.inputStage.stage — IXSCAN (index used) or COLLSCAN (full scan — BAD)
// executionStats.totalDocsExamined — documents scanned (lower = better)
// executionStats.executionTimeMillis — query time
```

## User Administration

### Create & Manage Users

```javascript
// Create admin user (must be run with --auth or on authenticated connection)
use admin
db.createUser({
  user: "admin",
  pwd: "securePassword123",
  roles: [
    { role: "userAdminAnyDatabase", db: "admin" },
    { role: "readWriteAnyDatabase", db: "admin" },
    { role: "clusterAdmin", db: "admin" }
  ]
})

// Create application user
use mydb
db.createUser({
  user: "appuser",
  pwd: "appPassword123",
  roles: [
    { role: "readWrite", db: "mydb" }
  ]
})

// Create read-only user
db.createUser({
  user: "readonly",
  pwd: "readOnlyPass",
  roles: [
    { role: "read", db: "mydb" }
  ]
})

// Update user roles
db.updateUser("appuser", {
  roles: [
    { role: "readWrite", db: "mydb" },
    { role: "read", db: "logs" }
  ]
})

// Change password
db.changeUserPassword("appuser", "newPassword123")

// List users
db.getUsers()
db.getUser("appuser")

// Remove user
db.dropUser("appuser")

// Authenticate
db.auth("admin", "securePassword123")
```

### Built-In Roles

| Role | Description |
|------|-------------|
| `read` | Read access to all non-system collections |
| `readWrite` | Read and write access |
| `dbAdmin` | Admin tasks (indexes, stats, repair) |
| `userAdmin` | Create and manage users/roles |
| `clusterAdmin` | Admin replica sets, sharding |
| `readAnyDatabase` | Read on all databases |
| `readWriteAnyDatabase` | ReadWrite on all databases |
| `userAdminAnyDatabase` | User admin on all databases |
| `dbAdminAnyDatabase` | DB admin on all databases |
| `root` | Superuser (all roles) |

## Replica Sets

### Initialize & Manage

```javascript
// Initialize replica set (run on primary node)
rs.initiate({
  _id: "myrs",
  members: [
    { _id: 0, host: "replica1:27017", priority: 2 },
    { _id: 1, host: "replica2:27017", priority: 1 },
    { _id: 2, host: "replica3:27017", priority: 1, arbiterOnly: true }
  ]
})

// Check replica set status
rs.status()
rs.conf()
rs.isMaster()                    // Check if current node is primary
rs.printReplicationInfo()        // Oplog info
rs.printSecondaryReplicationInfo()  // Replica lag

// Add/remove members
rs.add("new-replica:27017")
rs.remove("old-replica:27017")

// Reconfigure (force reconfig if primary is down)
cfg = rs.conf()
cfg.members[2].priority = 1
rs.reconfig(cfg, { force: true })

// Step down primary (force election)
rs.stepDown(60)                  // Step down for 60 seconds

// Freeze a node (prevent from becoming primary)
rs.freeze(60)                    // Freeze for 60 seconds
```

### Read Preference

## Backup & Restore

### mongodump / mongorestore

```bash
# Backup entire database
mongodump --db mydb --out /backup/

# Backup specific collection
mongodump --db mydb --collection users --out /backup/

# Backup all databases
mongodump --out /backup/

# Backup with authentication
mongodump -u admin -p password --authenticationDatabase admin --db mydb --out /backup/

# Backup remote server
mongodump --host primary:27017 --db mydb --out /backup/

# Compressed backup
mongodump --gzip --db mydb --out /backup/

# Oplog backup (point-in-time)
mongodump --oplog --out /backup/oplog/

# Restore entire database
mongorestore --db mydb /backup/mydb/

# Restore specific collection
mongorestore --db mydb --collection users /backup/mydb/users.bson

# Restore with drop (replace existing data)
mongorestore --drop --db mydb /backup/mydb/

# Restore compressed backup
mongorestore --gzip --db mydb /backup/mydb/

# Restore all databases
mongorestore /backup/
```

### MongoDB Database Tools (recommended)

```bash
# These are faster than mongodump/mongorestore
# Install: https://www.mongodb.com/try/download/database-tools

# Export to CSV/JSON
mongoexport --db mydb --collection users --type=csv --fields=name,email,age --out users.csv
mongoexport --db mydb --collection users --out users.json

# Import from CSV/JSON
mongoimport --db mydb --collection users --type=csv --headerline --file users.csv
mongoimport --db mydb --collection users --file users.json
```

```javascript
// Enable profiling (Level 0=off, 1=slow ops, 2=all ops)

db.system.profile.find().sort({ ts: -1 }).limit(10).pretty()

// Disable profiling
db.setProfilingLevel(0)

// Profile info
db.getProfilingStatus()
```

### Explain Plans

```javascript
// COLLSCAN = full collection scan (BAD — needs index)
// IXSCAN = index scan (GOOD)

// Key metrics:
// totalDocsExamined — lower is better
// totalKeysExamined — lower is better
// nReturned — documents returned
// executionTimeMillis — query time
```

## Useful Queries

### Schema Introspection

```javascript
// Database stats
db.stats()

// Collection stats
db.users.stats()

// List all collections with sizes
db.getCollectionNames().map(c => ({
  collection: c,
  size: db.getCollection(c).stats().size,
  count: db.getCollection(c).countDocuments()
}))

// Document schema analysis (MongoDB 3.6+)
db.users.aggregate([{ $schema: { content: { $value: "$$ROOT" } } }])

// Find documents with missing fields
db.users.find({ phone: { $exists: false } })

// Find documents with wrong type
db.users.find({ age: { $not: { $type: "number" } } })

// Document size distribution
db.users.aggregate([
  { $project: { size: { $bsonSize: "$$ROOT" } } },
  { $group: {
      _id: null,
      avgSize: { $avg: "$size" },
      maxSize: { $max: "$size" },
      minSize: { $min: "$size" }
  }}
])
```

### Change Streams

```javascript
// Watch for changes (requires replica set or sharded cluster)
const changeStream = db.users.watch()

changeStream.on("change", (change) => {
  console.log(change)
})

// Watch with filter
const changeStream = db.users.watch([
  { $match: { "operationType": "insert" } }
])

// Watch entire database
const changeStream = db.getDatabase("mydb").watch()

// Watch entire cluster
const changeStream = db.getMongo().watch()
```

## Troubleshooting

```bash
# Connection refused
sudo systemctl status mongod
sudo systemctl start mongod
sudo cat /var/log/mongodb/mongod.log | tail -50

# Authentication failed
mongosh -u admin -p password --authenticationDatabase admin
# Ensure auth is enabled in mongod.conf
# Ensure user exists in admin database

# Replica set issues
mongosh --eval "rs.status()"
# Check if primary is reachable from all secondaries
# Check clock synchronization (NTP)

# Disk space
db.stats().storageSize    # Current storage
du -sh /var/lib/mongodb/  # On disk

# Corrupt data
mongod --dbpath /var/lib/mongodb --repair

# Lock contention
db.currentOp({ "lockStats.Writing": { $gt: 0 } })

# OOM killed
# Check system logs
sudo journalctl -u mongod -n 100
# Increase WiredTiger cache or system RAM
```
